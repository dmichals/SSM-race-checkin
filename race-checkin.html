/** * SSM Race Check-In Sync Endpoint (Fixed & Concurrent-Safe) */ const SHEET_NAME = "RaceLog"; // <-- change to your sheet's tab name const HEADERS = [ "Sail #", "Boat Name", "Check-in Time", "Finish Time", "Start Area", "Weather Mark", "JAM Course", "Spinnaker Course", "Synced At" ]; const ROSTER_SHEET_NAME = "Boats"; // <-- change to your roster tab name const ROSTER_SAIL_COL = 1; // column A: Sail # const ROSTER_NAME_COL = 2; // column B: Boat Name const ROSTER_HEADER_ROWS = 1; // number of header rows to skip function doGet(e) { if (e && e.parameter && e.parameter.action === "roster") { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(ROSTER_SHEET_NAME); if (!sheet) { return responseJSON({ ok: false, error: "Roster sheet '" + ROSTER_SHEET_NAME + "' not found" }); } const lastRow = sheet.getLastRow(); const numRows = lastRow - ROSTER_HEADER_ROWS; let roster = []; if (numRows > 0) { const sailVals = sheet.getRange(ROSTER_HEADER_ROWS + 1, ROSTER_SAIL_COL, numRows, 1).getValues(); const nameVals = sheet.getRange(ROSTER_HEADER_ROWS + 1, ROSTER_NAME_COL, numRows, 1).getValues(); roster = sailVals .map((row, i) => ({ sail: String(row[0]).trim(), name: String(nameVals[i][0]).trim() })) .filter(b => b.sail !== ""); } return responseJSON({ ok: true, roster: roster }); } return responseJSON({ ok: false, error: "Unknown GET action" }); } function doPost(e) { // Prevent concurrent write overlaps const lock = LockService.getScriptLock(); const success = lock.tryLock(10000); // wait up to 10 seconds if (!success) { return responseJSON({ ok: false, error: "Server busy, lock timeout. Try again." }); } try { if (!e || !e.postData || !e.postData.contents) { return responseJSON({ ok: false, error: "Empty POST body" }); } const payload = JSON.parse(e.postData.contents); const records = payload.records || []; if (records.length === 0) { return responseJSON({ ok: true, written: 0 }); } const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(HEADERS); } else if (sheet.getLastRow() === 0) { sheet.appendRow(HEADERS); } const now = new Date().toISOString(); const rows = records.map(r => [ r.boat || "", r.boatName || "", r.checkinTime ? new Date(r.checkinTime).toLocaleString() : "", r.finishTime ? new Date(r.finishTime).toLocaleString() : "", r.startArea || "", r.weatherMark || "", r.jamCourse || "", r.spinCourse || "", now ]); sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, HEADERS.length).setValues(rows); return responseJSON({ ok: true, written: rows.length }); } catch (err) { return responseJSON({ ok: false, error: err.message }); } finally { lock.releaseLock(); } } /** * Helper to standardise output MIME types & bypass strict CORS issues */ function responseJSON(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.TEXT); }
